function MetroMap({ suburbs, onSelect }) {
  const [hovered, setHovered] = useState(null);

  return (
    <div className="relative">
      <svg viewBox="0 0 240 210" className="w-full">
        {/* Zone fills */}
        {ZONE_POLYGONS.map((z) => (
          <path key={z.zone} d={z.d} fill={z.color} stroke="#0e2408" strokeWidth="1.5" opacity="0.8" />
        ))}
        {/* Suburb dots */}
        {suburbs.map((s) => {
          const isHov = hovered === s.id;
          const pos = SUBURB_COORDS[s.id] || { x: 120, y: 105 };
          const x = pos.x;
          const y = pos.y;
          return (
            <g key={s.id}>
              <circle
                cx={x} cy={y} r={isHov ? 7 : 5}
                fill={isHov ? s.accent : s.color}
                stroke={isHov ? "#f5f0e4" : "#0e2408"}
                strokeWidth={isHov ? 1.5 : 1}
                style={{ cursor: "pointer", transition: "all 0.15s" }}
                onMouseEnter={() => setHovered(s.id)}
                onMouseLeave={() => setHovered(null)}
                onClick={() => onSelect(s)}
              />
              {isHov && (
                <text x={x} y={y - 10} textAnchor="middle" fontSize="7" fontFamily="'Work Sans',sans-serif" fill="#f5f0e4" fontWeight="600">

      </svg>
    </div>

function MapScreen({ onSelectSuburb }) {

  const [query, setQuery] = useState("");

                : { background: "transparent", color: "#5a7a3a", border: "1px solid #2d5016" }

                <p className="font-display text-lg font-semibold text-[#8ba86a]">{confidence}%</p>
              </div>
              <div className="absolute bottom-2 left-3">
                <p className="font-display text-base font-semibold text-[#f5f0e4]">{detected.name}</p>
                <p className="font-body text-[10px] italic text-[#8ba86a]">{detected.scientific}</p>
              </div>
            </div>
            <div className="p-3 flex gap-3">
              {[
                { label: "Rarity", value: rarityLabel[detected.rarity], color: rarityColor[detected.rarity] },
                { label: "Points", value: `+${detected.points}`, color: "#c8860a" },
                { label: "Type", value: detected.type, color: "#f5f0e4" },
              ].map(({ label, value, color }) => (
                <div key={label} className="flex-1 rounded-lg bg-[#0e2408] p-2 text-center">
                  <p className="font-mono text-[9px] text-[#5a7a3a] uppercase">{label}</p>
                  <p className="font-display text-sm font-semibold mt-0.5 capitalize" style={{ color }}>{value}</p>
                </div>
              ))}
            </div>
          </div>
        )}

        <div className="rounded-2xl p-4 space-y-3" style={{ background: "#1a3010", border: "1px solid #2d5016" }}>
          <p className="font-mono text-[9px] text-[#5a7a3a] uppercase tracking-widest">Field Record Template</p>
          {[
            { label: "Species", value: (detected && detected.name) || "Unknown" },
            { label: "Scientific Name", value: (detected && detected.scientific) || "—", italic: true },
            { label: "Date", value: dateStr },
            { label: "Time", value: timeStr },
            { label: "Suburb", value: (suburb && suburb.name) || "Kings Park" },
            { label: "Zone", value: (suburb && suburb.zone) || "Inner" },
            { label: "Recorder", value: USER.name },
          ].map(({ label, value, italic }) => (
            <div key={label} className="flex justify-between items-start gap-2 border-b border-[#2d5016] pb-2 last:border-0 last:pb-0">
              <span className="font-mono text-[10px] text-[#5a7a3a] uppercase tracking-wide flex-shrink-0">{label}</span>
              <span className={`font-body text-xs text-[#f5f0e4] text-right ${italic ? "italic" : ""}`}>{value}</span>
            </div>
          ))}
        </div>

        <div className="rounded-2xl p-4" style={{ background: "#1a3010", border: "1px solid #2d5016" }}>
          <label className="font-mono text-[9px] text-[#5a7a3a] uppercase tracking-widest block mb-2">Field Notes</label>
          <textarea
            value={notes}
            onChange={(e) => setNotes(e.target.value)}
            placeholder="Habitat, behaviour, notable features…"
            className="w-full bg-transparent font-body text-xs text-[#f5f0e4] placeholder-[#3d5a2a] resize-none outline-none leading-relaxed"
            rows={3}
          />
        </div>

        <button onClick={handleSave} className="w-full py-3.5 rounded-xl font-body font-semibold text-sm text-[#1e3a0f] bg-[#8ba86a]">
          Save Sighting · +{(detected && detected.points) || 0} pts
        </button>
        <button onClick={() => { setStep("scan"); setDetected(null); }} className="w-full py-2 rounded-xl font-body text-xs text-[#5a7a3a] bg-[#1e3a0f] border border-[#2d5016]">
          Retake Photo
        </button>
      </div>
    </div>
  );
}

// ── Root App ──────────────────────────────────────────────────────────────────

function App() {
  const [screen, setScreen] = useState("map");
  const [selectedSuburb, setSelectedSuburb] = useState(null);
  const [cameraSpecies, setCameraSpecies] = useState();
  const [cameraSuburb, setCameraSuburb] = useState();
  const [userLog, setUserLog] = useState([]);

  const handleSelectSuburb = (s) => {
    setSelectedSuburb(s);
    setScreen("region");
  };

  const handleLogSpecies = (sp) => {
    setCameraSpecies(sp);
    setCameraSuburb(selectedSuburb || undefined);
    setScreen("camera");
  };

  const handleSaveEntry = (entry) => {
    setUserLog((prev) => [entry, ...prev]);
  };

  const handleNav = (s) => {
    if (s !== "camera") { setCameraSpecies(undefined); setCameraSuburb(undefined); }
    setScreen(s);
  };

  const navScreen = screen === "region" ? "map" : screen;

  return (
    <div className="flex items-center justify-center w-full h-full bg-[#0a0a0a]">
      <div
        className="relative flex flex-col overflow-hidden"
        style={{ width: "min(390px, 100vw)", height: "min(844px, 100vh)", background: "#0e2408", borderRadius: "min(44px, 0px)", boxShadow: "0 40px 80px rgba(0,0,0,0.8)" }}
      >
        {/* Status bar */}
        <div className="absolute top-0 left-0 right-0 h-9 flex items-center justify-between px-6 z-20">
          <span className="font-mono text-[11px] text-[#f5f0e4] font-medium">9:41</span>
          <div className="flex items-center gap-1.5">
            <svg width="16" height="12" viewBox="0 0 16 12" fill="#f5f0e4"><rect x="0" y="4" width="3" height="8" rx="0.5" /><rect x="4.5" y="2.5" width="3" height="9.5" rx="0.5" /><rect x="9" y="0.5" width="3" height="11.5" rx="0.5" /><rect x="13.5" y="0" width="2.5" height="12" rx="0.5" opacity="0.3" /></svg>
            <svg width="20" height="12" viewBox="0 0 24 12" fill="none" stroke="#f5f0e4" strokeWidth="1.5"><rect x="0.75" y="0.75" width="19.5" height="10.5" rx="2" /><path d="M21.75 4v4a2 2 0 0 0 0-4z" fill="#f5f0e4" strokeWidth="0" /><rect x="2" y="2" width="14" height="8" rx="1" fill="#f5f0e4" /></svg>
          </div>
        </div>

        {/* Content */}
        <div className="absolute inset-0 top-0 bottom-[68px]">
          {screen === "map" && <MapScreen onSelectSuburb={handleSelectSuburb} />}
          {screen === "region" && selectedSuburb && (
            <SuburbScreen suburb={selectedSuburb} onBack={() => setScreen("map")} onLogSpecies={handleLogSpecies} />
          )}
          {screen === "camera" && <CameraScreen preselected={cameraSpecies} suburb={cameraSuburb} onSave={handleSaveEntry} />}
          {screen === "profile" && <ProfileScreen log={userLog} />}
          {screen === "resources" && <ResourcesScreen />}
        </div>

        <div className="absolute bottom-0 left-0 right-0">
          <BottomNav screen={navScreen} setScreen={handleNav} />
        </div>
      </div>
    </div>
  );
}

const root = ReactDOM.createRoot(document.getElementById("root"));
root.render(<App />);
